for script_name in  run_strong_dim128.sh run_weak_dim8_part224.sh run_weak_local_constant.sh run_dimension_sweep.sh; do
    
    # Modifica automatica dei tuoi script per forzare l'ancoraggio ai core fisici
    sed -i 's/mpirun -np/mpirun --map-by core --bind-to core -np/g' "$script_name"
    
    # Crea il nome del file PBS (max 15 caratteri)
    job_file="job_${script_name%.sh}.pbs"
    job_name=$(echo "${script_name%.sh}" | cut -c 1-15)
    
    cat > "$job_file" << EOF
#!/bin/bash
#PBS -N $job_name
#PBS -q cpu
#PBS -l select=1:ncpus=28:mpiprocs=28
#PBS -l walltime=15:59:59
#PBS -j oe

# Carica l'ambiente Spack e MPI in modo che coincida con la compilazione
source /software/spack/share/spack/setup-env.sh
spack load gcc@15.2.0
spack load openmpi@5.0.8

# Spostati nella cartella in cui ti trovi adesso
cd "$PWD"

# Dai i permessi ed esegui lo script
chmod +x "$script_name"
./"$script_name"
EOF

    # Sottometti il job alla coda in modalità batch [cite: 215-216]
    echo "Sottomissione di $job_file in corso..."
    qsub "$job_file"
done
